cmake_minimum_required(VERSION 3.16)
project(petscop_c89 LANGUAGES C)
set(CMAKE_C_STANDARD 90)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(SDL2 REQUIRED CONFIG)
find_package(OpenGL REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(petscop_c89 src/engine.c src/game.c src/json.c src/lighting.c src/physics.c src/localization.c)
target_link_libraries(petscop_c89 PRIVATE SDL2::SDL2 OpenGL::GL ZLIB::ZLIB)
if(NOT WIN32)
  target_link_libraries(petscop_c89 PRIVATE m)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(petscop_c89 PRIVATE -Wall -Wextra -Wshadow -Wconversion -pedantic-errors)
endif()
